🌾 Farmer Action Checklist based on Water Quality

pH < 6.0 (Acidic Hazard) Risk of root damage and heavy metal toxicity. Add agricultural lime or potassium bicarbonate to reservoir.
pH > 8.0 (Alkaline Lockout) Phosphorus and micronutrients become locked in water. Add diluted citric or phosphoric acid.
DO < 5.0 mg/L (Hypoxia) Stagnant water causes root asphyxiation. Run irrigation water pump or bubbler for 15 minutes to aerate.
TDS > 800 ppm (High Salts) Risk of salt crusting and leaf burn. Dilute reservoir with fresh rainwater or canal supply.

🌱 Farmer Irrigation & Moisture Guide

Smart watering practices tailored for Indian agriculture and sustainable water conservation

🌾 Rice / Paddy Field

Maintain 2-5 cm shallow standing water during tillering. Drain field 10 days before harvest to save up to 30% groundwater.

🌿 Wheat Crop

Requires 4-6 critical irrigations: Crown root initiation (21 days), Tillering, Boot stage, and Grain filling stage.

🍅 Vegetables & Horticulture

Use drip irrigation. Irrigate in early morning (6 AM - 8 AM) or late evening to minimize midday evaporation losses.

💻 Arduino Uno Firmware (FarmerWaterMonitor.ino)

Upload this sketch to your Arduino Uno board using the Arduino IDE

/*
 * KISAN MITRA - Smart Farmer Water Quality & Irrigation Station
 * Board: Arduino Uno R3 | Baud: 115200 (Web Serial Compatible)
 */
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int PIN_PH    = A0;
const int PIN_TDS   = A1;
const int PIN_DO    = A2;
const int PIN_LEVEL = A3;
const int PIN_RELAY = 8; // Active LOW

float phVal = 7.20;
float tdsVal = 320.0;
float doVal = 7.40;
int   levelPct = 68;
bool  pumpActive = false;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY, OUTPUT);
  digitalWrite(PIN_RELAY, HIGH); // Pump OFF

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("KISAN MITRA AGRI");
  lcd.setCursor(0, 1);
  lcd.print("SMART WATER v2.0");
  delay(1500);
}

void loop() {
  // Read analog pins
  int rawPH = analogRead(PIN_PH);
  phVal = 7.00 + ((2.50 - (rawPH * 5.0 / 1023.0)) * 3.5);

  int rawTDS = analogRead(PIN_TDS);
  tdsVal = (rawTDS * 5.0 / 1023.0) * 300.0;

  int rawDO = analogRead(PIN_DO);
  doVal = (rawDO * 5.0 / 1023.0) * 5.0;

  int rawLvl = analogRead(PIN_LEVEL);
  levelPct = map(rawLvl, 100, 750, 0, 100);

  // Auto pump logic
  if (levelPct < 30 && !pumpActive) {
    pumpActive = true;
    digitalWrite(PIN_RELAY, LOW); // Turn pump ON
  } else if (levelPct > 85 && pumpActive) {
    pumpActive = false;
    digitalWrite(PIN_RELAY, HIGH); // Turn pump OFF
  }

  // Send JSON over USB Serial to Kisan Mitra web dashboard
  Serial.print("{\"ph\":"); Serial.print(phVal, 2);
  Serial.print(",\"tds\":"); Serial.print((int)tdsVal);
  Serial.print(",\"do\":"); Serial.print(doVal, 1);
  Serial.print(",\"level\":"); Serial.print(levelPct);
  Serial.print(",\"pump\":"); Serial.print(pumpActive ? "true" : "false");
  Serial.println("}");

  // Update physical 16x2 LCD
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("pH:"); lcd.print(phVal, 1);
  lcd.print(" DO:"); lcd.print(doVal, 1);
  lcd.setCursor(0, 1);
  lcd.print("TDS:"); lcd.print((int)tdsVal);
  lcd.print(" L:"); lcd.print(levelPct); lcd.print("%");

  delay(1000);
}
